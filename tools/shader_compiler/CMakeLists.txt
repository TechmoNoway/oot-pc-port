cmake_minimum_required(VERSION 3.24)
project(oot_shader_tool LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT OOT_SHADER_VENDOR OR NOT OOT_SPIRV_HEADERS)
  message(FATAL_ERROR "Set the verified compiler vendor and SPIR-V header directories")
endif()
file(TO_CMAKE_PATH "${OOT_SHADER_VENDOR}" OOT_SHADER_VENDOR)
file(TO_CMAKE_PATH "${OOT_SPIRV_HEADERS}" OOT_SPIRV_HEADERS)
set(DEP "${OOT_SHADER_VENDOR}/gpu/vk")
file(GLOB_RECURSE IR_SOURCES CONFIGURE_DEPENDS "${DEP}/eden/shader_recompiler/*.cpp")
file(GLOB_RECURSE SIRIT_SOURCES CONFIGURE_DEPENDS "${DEP}/sirit/src/*.cpp")
add_library(oot_shader_dependency STATIC ${IR_SOURCES} ${SIRIT_SOURCES})
target_include_directories(oot_shader_dependency PUBLIC "${DEP}/compat" "${DEP}/eden"
  "${DEP}/sirit/include" "${OOT_SHADER_VENDOR}/gpu/compat" "${OOT_SPIRV_HEADERS}/include")
target_include_directories(oot_shader_dependency PRIVATE "${DEP}/sirit/src")
target_compile_definitions(oot_shader_dependency PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN GPU_SHADER_LOG=1)
if(MSVC)
  target_compile_options(oot_shader_dependency PRIVATE /bigobj /EHsc /Zc:__cplusplus /Zc:preprocessor /W0)
  set_source_files_properties("${DEP}/eden/shader_recompiler/frontend/maxwell/translate/impl/internal_stage_buffer_entry_read.cpp"
    PROPERTIES COMPILE_OPTIONS "/FI${CMAKE_CURRENT_SOURCE_DIR}/msvc_compat.h")
endif()
add_executable(oot-shader-tool main.cpp)
target_link_libraries(oot-shader-tool PRIVATE oot_shader_dependency)
if(MSVC)
  target_compile_options(oot-shader-tool PRIVATE /W4 /EHsc /Zc:__cplusplus /Zc:preprocessor)
endif()
